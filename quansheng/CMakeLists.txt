cmake_minimum_required(VERSION 3.16)
project(QuanshengDockLinux VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QDOCK_SERIAL "Build serial-port support (requires Qt6 SerialPort)" ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Widgets)
add_library(qdock-core src/core/parser.cpp src/core/displaymodel.cpp
  src/experimental/rssiquery.cpp src/experimental/registerquery.cpp
  src/experimental/eepromquery.cpp src/experimental/keycontrol.cpp)
target_include_directories(qdock-core PUBLIC src)
target_compile_options(qdock-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(qdock-probe src/main.cpp)
add_library(qdock-json src/eventjson.cpp)
target_link_libraries(qdock-json PUBLIC qdock-core Qt6::Core)
target_link_libraries(qdock-probe PRIVATE qdock-json)
add_executable(qdock-server src/lan/server.cpp)
target_link_libraries(qdock-server PRIVATE qdock-json Qt6::Network)
target_compile_options(qdock-server PRIVATE -Wall -Wextra -Wpedantic)
add_custom_target(qdock-build-timestamp
  COMMAND ${CMAKE_COMMAND}
    -DOUTPUT_FILE=${CMAKE_CURRENT_BINARY_DIR}/qdock_build_timestamp.h
    -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/build-timestamp.cmake
  BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/qdock_build_timestamp.h
  COMMENT "Actualizando fecha y hora de compilación de qdock-server")
add_dependencies(qdock-server qdock-build-timestamp)
target_include_directories(qdock-server PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(qdock-probe PRIVATE -Wall -Wextra -Wpedantic)
if(QDOCK_SERIAL)
  find_package(Qt6 REQUIRED COMPONENTS SerialPort)
  add_library(qdock-serial src/serial/reader.cpp)
  target_include_directories(qdock-serial PUBLIC src)
  target_link_libraries(qdock-serial PUBLIC Qt6::SerialPort)
  target_link_libraries(qdock-probe PRIVATE qdock-serial)
  target_compile_definitions(qdock-probe PRIVATE QDOCK_SERIAL)
  target_sources(qdock-server PRIVATE src/lan/serialsource.cpp src/lan/serialsource.h)
  target_link_libraries(qdock-server PRIVATE qdock-serial)
  target_compile_definitions(qdock-server PRIVATE QDOCK_SERIAL)
  set_target_properties(qdock-server PROPERTIES AUTOMOC ON)
  add_executable(qdock-server-gui src/gui/serverwindow.cpp)
  target_link_libraries(qdock-server-gui PRIVATE Qt6::Widgets Qt6::SerialPort)
  target_compile_options(qdock-server-gui PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(qdock-rssi-query
    src/experimental/rssi_main.cpp)
  target_include_directories(qdock-rssi-query PRIVATE src)
  target_link_libraries(qdock-rssi-query PRIVATE qdock-core Qt6::Core Qt6::SerialPort)
  target_compile_options(qdock-rssi-query PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(qdock-register-query
    src/experimental/register_main.cpp)
  target_include_directories(qdock-register-query PRIVATE src)
  target_link_libraries(qdock-register-query PRIVATE qdock-core Qt6::Core Qt6::SerialPort)
  target_compile_options(qdock-register-query PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(qdock-eeprom-query
    src/experimental/eeprom_main.cpp)
  target_include_directories(qdock-eeprom-query PRIVATE src)
  target_link_libraries(qdock-eeprom-query PRIVATE qdock-core Qt6::Core Qt6::SerialPort)
  target_compile_options(qdock-eeprom-query PRIVATE -Wall -Wextra -Wpedantic)
endif()

  endif()
