cmake_minimum_required(VERSION 3.16)
project(QuanshengDockLinux VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QDOCK_SERIAL "Build serial-port support (requires Qt6 SerialPort)" ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(qdock-core src/core/parser.cpp)
target_include_directories(qdock-core PUBLIC src)
target_compile_options(qdock-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(qdock-probe src/main.cpp)
target_link_libraries(qdock-probe PRIVATE qdock-core Qt6::Core)
target_compile_options(qdock-probe PRIVATE -Wall -Wextra -Wpedantic)
if(QDOCK_SERIAL)
  find_package(Qt6 REQUIRED COMPONENTS SerialPort)
  target_sources(qdock-probe PRIVATE src/serial/reader.cpp)
  target_link_libraries(qdock-probe PRIVATE Qt6::SerialPort)
  target_compile_definitions(qdock-probe PRIVATE QDOCK_SERIAL)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(parser-tests tests/parser_tests.cpp)
  target_link_libraries(parser-tests PRIVATE qdock-core)
  add_test(NAME parser COMMAND parser-tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME probe-cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/probe_test.py $<TARGET_FILE:qdock-probe>)
  if(QDOCK_SERIAL)
    add_test(NAME serial-read-only COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/serial_test.py $<TARGET_FILE:qdock-probe>)
  endif()
endif()
